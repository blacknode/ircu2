# ---------------------------------------------------------------------------
# Generated source: the configuration file parser
# ---------------------------------------------------------------------------
find_package(BISON REQUIRED)

bison_target(IrcdParser
  ircd_parser.y
  "${CMAKE_CURRENT_BINARY_DIR}/ircd_parser.c"
  DEFINES_FILE "${CMAKE_CURRENT_BINARY_DIR}/ircd_parser.h")

# ---------------------------------------------------------------------------
# Generated source: the character attribute table
#
# table_gen is a build-time helper: it prints the table that ircd_string.c
# #includes.  It never gets installed.
# ---------------------------------------------------------------------------
add_executable(table_gen table_gen.c)
target_link_libraries(table_gen PRIVATE ircu_config)

set(IRCU_CHATTR_TAB "${CMAKE_CURRENT_BINARY_DIR}/chattr.tab.c")
add_custom_command(
  OUTPUT "${IRCU_CHATTR_TAB}"
  COMMAND "${CMAKE_COMMAND}"
          -DIRCU_COMMAND=$<TARGET_FILE:table_gen>
          -DIRCU_OUTPUT=${IRCU_CHATTR_TAB}
          -P "${PROJECT_SOURCE_DIR}/cmake/RunAndCapture.cmake"
  DEPENDS table_gen "${PROJECT_SOURCE_DIR}/cmake/RunAndCapture.cmake"
  COMMENT "Generating chattr.tab.c"
  VERBATIM)
add_custom_target(ircu_chattr DEPENDS "${IRCU_CHATTR_TAB}")

# ircd_string.c #includes chattr.tab.c rather than linking against it, so the
# dependency has to be spelled out.  ircd/test compiles the same file and
# repeats this; source properties do not cross directories.
set_source_files_properties(ircd_string.c
  PROPERTIES OBJECT_DEPENDS "${IRCU_CHATTR_TAB}")

# Let ircd/test reuse both.
set(IRCU_CHATTR_TAB "${IRCU_CHATTR_TAB}" PARENT_SCOPE)

# ---------------------------------------------------------------------------
# Generated header: the applied-patch list
#
# ircd-patch writes include/patchlist.h into the source tree.  When it has
# never been run we supply an empty one from a build-tree directory that is
# searched *after* include/, so a real patchlist.h always wins.
# ---------------------------------------------------------------------------
set(IRCU_PATCHLIST_DIR "${CMAKE_CURRENT_BINARY_DIR}/patchlist-default")
if(NOT EXISTS "${IRCU_PATCHLIST_DIR}/patchlist.h")
  file(WRITE "${IRCU_PATCHLIST_DIR}/patchlist.h"
    "/* This file was automatically generated by the build system */\n"
    "#define PATCHLIST \"\"\n")
endif()

# ---------------------------------------------------------------------------
# Generated source: version.c
#
# Reproduces what ircd/version.c.SH used to do: bump the build generation
# counter, stamp the build time and list an MD5 of every source file.  It is
# regenerated whenever any of those sources change.
# ---------------------------------------------------------------------------
file(GLOB IRCU_VERSION_INPUTS CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.l"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.y"
  "${PROJECT_SOURCE_DIR}/include/*.h")

set(IRCU_VERSION_C "${CMAKE_CURRENT_BINARY_DIR}/version.c")
add_custom_command(
  OUTPUT "${IRCU_VERSION_C}"
  COMMAND "${CMAKE_COMMAND}"
          -DIRCU_SOURCE_DIR=${PROJECT_SOURCE_DIR}
          -DIRCU_OUTPUT=${IRCU_VERSION_C}
          -P "${PROJECT_SOURCE_DIR}/cmake/GenerateVersion.cmake"
  DEPENDS ${IRCU_VERSION_INPUTS}
          "${PROJECT_SOURCE_DIR}/cmake/GenerateVersion.cmake"
  COMMENT "Generating version.c"
  VERBATIM)

# The event engines that survived configuration, plus the selected TLS
# backend.  See cmake/IrcuOptions.cmake and cmake/IrcuTLS.cmake.
list(APPEND IRCU_SOURCES ${IRCU_ENGINE_SOURCES} ${IRCU_TLS_SOURCE})

add_executable(ircd
  ${IRCU_SOURCES}
  ${BISON_IrcdParser_OUTPUTS}
  "${IRCU_VERSION_C}")

# Order matters: include/ has to come before the patchlist fallback, and
# listing it explicitly here puts it ahead of the copy ircu_config carries.
target_include_directories(ircd PRIVATE
  "${CMAKE_CURRENT_BINARY_DIR}"          # ircd_parser.h, chattr.tab.c
  "${PROJECT_SOURCE_DIR}/include"        # a patchlist.h written by ircd-patch
  "${IRCU_PATCHLIST_DIR}")               # ... otherwise an empty one
target_link_libraries(ircd PRIVATE
  ircu_config ircu::tls ${IRCU_SYSTEM_LIBRARIES} ${CMAKE_DL_LIBS})

# Modules resolve the core's symbols against the executable itself, so the
# ircd has to carry a dynamic symbol table (-rdynamic).  See doc/readme.modules.
set_target_properties(ircd PROPERTIES ENABLE_EXPORTS ON)

add_dependencies(ircd ircu_chattr)

# ---------------------------------------------------------------------------
# Helper programs
# ---------------------------------------------------------------------------
add_executable(umkpasswd
  ircd_alloc.c
  ircd_crypt_native.c
  ircd_crypt_plain.c
  ircd_crypt_smd5.c
  ircd_md5.c
  ircd_string.c
  memdebug.c
  umkpasswd.c)
target_include_directories(umkpasswd PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(umkpasswd PRIVATE ircu_config ${IRCU_SYSTEM_LIBRARIES})
add_dependencies(umkpasswd ircu_chattr)

add_executable(convert-conf convert-conf.c)
target_link_libraries(convert-conf PRIVATE ircu_config ${IRCU_SYSTEM_LIBRARIES})

# ---------------------------------------------------------------------------
# Installation
#
# ircu installs more than plain binaries: the data directory has to exist with
# the right ownership, the binary is installed under a timestamped name with a
# symlink pointing at it, and example.conf is seeded into the data directory.
# ---------------------------------------------------------------------------
# The hooks need values known only at configure time, so they go through
# generated scripts.  install() rules run in the order they are declared: the
# pre-install script clears a previous install's symlink out of the way, then
# the binaries land, then the post-install script does the fixups.
configure_file(
  "${PROJECT_SOURCE_DIR}/cmake/PreInstallHooks.cmake.in"
  "${PROJECT_BINARY_DIR}/cmake/PreInstallHooks.cmake"
  @ONLY)
configure_file(
  "${PROJECT_SOURCE_DIR}/cmake/InstallHooks.cmake.in"
  "${PROJECT_BINARY_DIR}/cmake/InstallHooks.cmake"
  @ONLY)

install(SCRIPT "${PROJECT_BINARY_DIR}/cmake/PreInstallHooks.cmake")

install(TARGETS ircd umkpasswd convert-conf
  RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")

install(SCRIPT "${PROJECT_BINARY_DIR}/cmake/InstallHooks.cmake")
